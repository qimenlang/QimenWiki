## 3. Compute Shader实现方案

### 3.1 主调度Shader
```glsl
#version 450 core
layout(local_size_x = 16, local_size_y = 16) in;
layout(rgba32f, binding = 0) uniform image2D outputImage;

// SSBO绑定
layout(std430, binding = 0) buffer Triangles { Triangle triangles[]; };
layout(std430, binding = 1) buffer Materials { Material materials[]; };
layout(std430, binding = 2) buffer BVHNodes { BVHNode bvh_nodes[]; };

void main() {
    ivec2 pixel_coord = ivec2(gl_GlobalInvocationID.xy);
    ivec2 image_size = imageSize(outputImage);
    
    if (pixel_coord.x >= image_size.x || pixel_coord.y >= image_size.y) 
        return;
    
    // 光线追踪主循环
    vec3 color = traceRay(pixel_coord, image_size);
    
    imageStore(outputImage, pixel_coord, vec4(color, 1.0));
}
```

### 3.2 光线生成
```glsl
Ray generateRay(ivec2 pixel, ivec2 resolution) {
    // 相机参数
    vec3 camera_pos = vec3(0, 1, 5);
    vec3 look_at = vec3(0, 0, 0);
    vec3 up = vec3(0, 1, 0);
    float fov = 45.0;
    
    // 计算相机坐标系
    vec3 w = normalize(camera_pos - look_at);
    vec3 u = normalize(cross(up, w));
    vec3 v = cross(w, u);
    
    // 生成光线
    float aspect = float(resolution.x) / float(resolution.y);
    vec2 uv = (vec2(pixel) + 0.5) / vec2(resolution);
    uv = uv * 2.0 - 1.0;
    uv.x *= aspect;
    
    vec3 ray_dir = normalize(uv.x * u + uv.y * v - w);
    
    return Ray(camera_pos, ray_dir, 0.001, 1000.0);
}
```

### 3.3 BVH遍历与相交测试
```glsl
bool intersectBVH(Ray ray, inout HitRecord hit) {
    int stack[64];
    int stack_ptr = 0;
    stack[stack_ptr++] = 0; // 根节点
    
    bool hit_anything = false;
    float closest_so_far = ray.t_max;
    
    while (stack_ptr > 0) {
        int node_idx = stack[--stack_ptr];
        BVHNode node = bvh_nodes[node_idx];
        
        if (!intersectAABB(ray, node.min_bound, node.max_bound))
            continue;
            
        if (node.triangle_count > 0) {
            // 叶子节点，测试三角形
            for (int i = 0; i < node.triangle_count; i++) {
                Triangle tri = triangles[node.triangle_offset + i];
                if (intersectTriangle(ray, tri, closest_so_far, hit)) {
                    hit_anything = true;
                    closest_so_far = hit.t;
                }
            }
        } else {
            // 内部节点，推入子节点
            if (node.left_child >= 0) stack[stack_ptr++] = node.left_child;
            if (node.right_child >= 0) stack[stack_ptr++] = node.right_child;
        }
    }
    
    return hit_anything;
}
```

### 3.4 材质系统
```glsl
vec3 sampleMaterial(Ray ray, HitRecord hit, inout vec3 attenuation, inout Ray scattered) {
    Material mat = materials[hit.material_id];
    
    if (mat.emission.r > 0.0 || mat.emission.g > 0.0 || mat.emission.b > 0.0) {
        return mat.emission; // 自发光材质
    }
    
    if (random() < mat.metallic) {
        // 金属材质
        vec3 reflected = reflect(ray.direction, hit.normal);
        scattered = Ray(hit.position, reflected + mat.roughness * randomInUnitSphere());
        attenuation = mat.albedo;
        return vec3(0.0);
    } else if (random() < 0.8) {
        // 漫反射材质
        vec3 target = hit.normal + randomInUnitSphere();
        scattered = Ray(hit.position, target);
        attenuation = mat.albedo;
        return vec3(0.0);
    } else {
        // 折射材质
        float refraction_ratio = hit.front_face ? (1.0 / mat.ior) : mat.ior;
        vec3 unit_direction = normalize(ray.direction);
        vec3 refracted = refract(unit_direction, hit.normal, refraction_ratio);
        scattered = Ray(hit.position, refracted);
        attenuation = vec3(1.0);
        return vec3(0.0);
    }
}
```

### 3.5 路径追踪核心
```glsl
vec3 traceRay(ivec2 pixel, ivec2 resolution) {
    Ray ray = generateRay(pixel, resolution);
    vec3 color = vec3(0.0);
    vec3 throughput = vec3(1.0);
    
    for (int depth = 0; depth < MAX_DEPTH; depth++) {
        HitRecord hit;
        
        if (!intersectBVH(ray, hit)) {
            // 击中天空盒
            color += throughput * sampleSkybox(ray.direction);
            break;
        }
        
        Ray scattered;
        vec3 attenuation;
        vec3 emitted = sampleMaterial(ray, hit, attenuation, scattered);
        
        color += throughput * emitted;
        throughput *= attenuation;
        
        // 俄罗斯轮盘赌终止
        if (depth > 3) {
            float continue_probability = min(max(throughput.x, max(throughput.y, throughput.z)), 1.0);
            if (random() > continue_probability)
                break;
            throughput /= continue_probability;
        }
        
        ray = scattered;
    }
    
    return color;
}
```

### 4.1 内存优化

- 压缩BVH节点(8字节对齐)
- 纹理缓存友好访问模式

### 4.2 计算优化
```glsl
// 提前终止优化
bool intersectAABB(Ray ray, vec3 min_bound, vec3 max_bound) {
    vec3 inv_dir = 1.0 / ray.direction;
    vec3 t0 = (min_bound - ray.origin) * inv_dir;
    vec3 t1 = (max_bound - ray.origin) * inv_dir;
    
    vec3 tmin = min(t0, t1);
    vec3 tmax = max(t0, t1);
    
    float t_min = max(max(tmin.x, tmin.y), tmin.z);
    float t_max = min(min(tmax.x, tmax.y), tmax.z);
    
    return t_max >= max(t_min, 0.0);
}
```

### 4.3 降噪处理
```glsl
// 时域累积
layout(rgba32f, binding = 1) uniform image2D accumulationBuffer;

vec3 temporalAccumulation(ivec2 pixel, vec3 current_color) {
    vec3 accumulated = imageLoad(accumulationBuffer, pixel).rgb;
    float frame_count = accumulated.a;
    
    float blend_factor = 1.0 / (frame_count + 1.0);
    vec3 new_color = mix(accumulated.rgb, current_color, blend_factor);
    
    imageStore(accumulationBuffer, pixel, vec4(new_color, frame_count + 1.0));
    return new_color;
}
```

### 5.2 渲染循环
```cpp
void RayTracingRenderer::render() {
    glUseProgram(computeProgram);
    
    // 绑定SSBOs
    glBindBufferBase(GL_SHADER_STORAGE_BUFFER, 0, triangleSSBO);
    glBindBufferBase(GL_SHADER_STORAGE_BUFFER, 1, materialSSBO);
    glBindBufferBase(GL_SHADER_STORAGE_BUFFER, 2, bvhSSBO);
    
    // 绑定输出纹理
    glBindImageTexture(0, outputTexture, 0, GL_FALSE, 0, GL_WRITE_ONLY, GL_RGBA32F);
    glBindImageTexture(1, accumulationTexture, 0, GL_FALSE, 0, GL_READ_WRITE, GL_RGBA32F);
    
    // 分派计算着色器
    ivec2 workGroups = (imageSize + 15) / 16;
    glDispatchCompute(workGroups.x, workGroups.y, 1);
    
    glMemoryBarrier(GL_SHADER_IMAGE_ACCESS_BARRIER_BIT);
}
```
